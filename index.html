-- Gegenereerd uit Enterprise Architect
CREATE TABLE persoon (
  id             INTEGER PRIMARY KEY,
  voornamen      VARCHAR(200),
  geslachtsnaam  VARCHAR(200),
  geboortedatum  DATE,
  adres_id       INTEGER
    REFERENCES adres(id)
);

CREATE TABLE adres (
  id             INTEGER PRIMARY KEY,
  straatnaam     VARCHAR(255),
  postcode       CHAR(6)
);
// dbdiagram.io
Table persoon {
  id             integer [pk]
  voornamen      varchar
  geslachtsnaam  varchar
  geboortedatum  date
  adres_id       integer [ref: > adres.id]
}

Table adres {
  id             integer [pk]
  straatnaam     varchar
  postcode       char(6)
}
@startuml
class Persoon {
  voornamen : String
  geslachtsnaam : String
  geboortedatum : Date
}
class Adres {
  straatnaam : String
  postcode : String
}
Persoon "0..*" --> "1" Adres : woont op
@enduml
:Persoon a owl:Class ;
  rdfs:label "Persoon"@nl .

:geboortedatum a owl:DatatypeProperty ;
  rdfs:domain :Persoon ;
  rdfs:range  xsd:date .

:woontOp a owl:ObjectProperty ;
  rdfs:domain :Persoon ;
  rdfs:range  :Adres .
